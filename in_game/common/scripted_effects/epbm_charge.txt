# EPBM — display snapshot. Estates use variable maps. Modifier snapshots
# use named vars for the known 7 + "other" bucket. The crown has no
# snapshot: the engine's building maintenance line carries its share.

# ═══════════════════════════════════════════════
# Snapshot pay vars into show vars so tooltips keep displaying last
# month's charged amount after the next recalc overwrites pay vars.
# The auto_modifier scales read the show vars, so the AI must run this
# after every calculation too. Scope: country.
# ═══════════════════════════════════════════════

epbm_snapshot_for_display = {
	epbm_snapshot_estates_dynamic = yes
	epbm_for_each_estate = { effect = epbm_snapshot_mod }

	if = {
		limit = { has_variable = epbm_mod_other }
		set_variable = { name = epbm_show_mod_other value = var:epbm_mod_other }
	}
}
