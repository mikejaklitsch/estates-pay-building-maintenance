# Sum of all non-estate building_upkeep_multiplier sources.
# Mirrors exactly what the engine accumulates from country_base_values
# (0.2 vanilla + 0.8 inject = 1.0), game rule offsets, and yearly creep.
# The creep adds to the base; it does not scale it. Used both for
# auto_modifier scales and for charge_factor.

epbm_base_multiplier = {
	value = 1.0

	if      = { limit = { has_game_rule = epbm_baseline_50 } subtract = 0.5 }
	else_if = { limit = { has_game_rule = epbm_baseline_20 } subtract = 0.8 }

	if = {
		limit = { NOT = { has_game_rule = epbm_yearly_creep_disabled } }
		add = {
			value = current_year
			subtract = 1337
			multiply = 0.001
		}
	}
}

# GUI display script values. Evaluated from estate scope:
# estate -> owner (country) -> read variable map with prev (estate) as key.

	owner = {
		if = {
			limit = { has_variable_map = epbm_prev is_key_in_variable_map = { name = epbm_prev target = prev } }
			value = "variable_map(epbm_prev|prev)"
		}
	}
